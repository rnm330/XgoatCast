cmake_minimum_required(VERSION 3.20)
project(XgoatCast LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Use an x64 MSVC Windows toolchain")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The bundled Agora SDK requires an x64 build")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(AGORA_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/deps/agora/sdk" CACHE PATH "Agora SDK directory")
add_executable(XgoatCast WIN32 main.cpp audio.cpp session.cpp core.h version.h app.rc)
target_compile_definitions(XgoatCast PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(XgoatCast PRIVATE /utf-8 /W4 /EHsc)
target_link_options(XgoatCast PRIVATE /MANIFEST:NO)
target_include_directories(XgoatCast PRIVATE deps "${AGORA_ROOT}/high_level_api/include")
target_link_libraries(XgoatCast PRIVATE "${AGORA_ROOT}/x86_64/agora_rtc_sdk.dll.lib" winhttp dwmapi gdiplus d2d1 dwrite comctl32 shell32 shlwapi ole32 oleaut32 uuid winmm mmdevapi advapi32 user32 gdi32)
